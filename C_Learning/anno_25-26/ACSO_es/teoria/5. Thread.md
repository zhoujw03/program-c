# 线程（Thread）笔记（含 POSIX semaphore + macOS GCD 双版本）

# 一、线程是什么
**线程（Thread）** 是进程内部的执行流，是操作系统调度的最小单位。

特点：
- 一个进程可以包含多个线程
- 所有线程**共享全局变量、文件描述符、堆内存**
- 每个线程有**独立的栈空间**
- 线程创建、销毁、切换比进程快得多
- 多线程用于并发执行、提高效率

---

# 二、线程核心API（pthread）
头文件：
```c
#include <pthread.h>
```

核心函数：
- `pthread_create`：创建线程
- `pthread_join`：等待线程结束（阻塞）
- `pthread_exit`：线程主动退出
- `pthread_self`：获取当前线程ID

编译必须加：
```bash
gcc main.c -o main -lpthread
```

# 三、例子1：创建简单线程
```c
#include <stdio.h>
#include <pthread.h>

// 线程执行函数
// 参数：void*  返回值：void*
void* thread_func(void *arg) {
    printf("线程运行中\n");
    pthread_exit(NULL); // 线程退出
}

int main() {
    pthread_t t;

    // 创建线程
    // 参数：线程ID、属性（NULL默认）、函数名、参数
    pthread_create(&t, NULL, thread_func, NULL);

    // 等待线程结束
    pthread_join(t, NULL);

    return 0;
}
```

# 四、例子2：线程传递数据
```c
#include <stdio.h>
#include <pthread.h>

void* func(void *arg) {
    // 将void* 转换回 int*
    int value = *(int*)arg;
    printf("线程收到数据：%d\n", value);
    return NULL;
}

int main() {
    pthread_t t;
    int x = 123;

    // 传递变量x的地址
    pthread_create(&t, NULL, func, &x);

    pthread_join(t, NULL);
    return 0;
}
```

# 五、线程共享资源问题（竞态条件）
多个线程同时修改同一个变量 → **结果错误、乱序、不可预测**

解决方法：
1. **互斥锁 Mutex**：保护共享变量
2. **信号量 Semaphore**：控制执行顺序

---

# 六、互斥锁 Mutex
作用：**同一时间只允许一个线程进入临界区**

```c
#include <stdio.h>
#include <pthread.h>

int counter = 0;                // 共享变量
pthread_mutex_t m;               // 互斥锁

void* func(void *arg) {
    for (int i = 0; i < 10000; i++) {
        pthread_mutex_lock(&m);     // 加锁（进入临界区）
        counter++;                  // 安全修改共享变量
        pthread_mutex_unlock(&m);   // 解锁（离开临界区）
    }
    return NULL;
}

int main() {
    pthread_t t1, t2;
    pthread_mutex_init(&m, NULL); // 初始化锁

    pthread_create(&t1, NULL, func, NULL);
    pthread_create(&t2, NULL, func, NULL);

    pthread_join(t1, NULL);
    pthread_join(t2, NULL);

    printf("counter = %d\n", counter); // 结果一定是 20000
    pthread_mutex_destroy(&m); // 销毁锁
    return 0;
}
```

# 七、信号量 Semaphore（POSIX sem_init 标准版本）
⚠️ 说明：macOS 标记 `sem_init` 为废弃，但**考试/作业必须会**

头文件：
```c
#include <semaphore.h>
```

核心函数：
- `sem_init(&sem, 0, value)`：初始化（0=线程共享）
- `sem_wait(&sem)`：P操作，申请资源（-1）
- `sem_post(&sem)`：V操作，释放资源（+1）
- `sem_destroy(&sem)`：销毁信号量

---

# 八、例子：线程交替执行 inc/dec（sem_init 版本）
```c
#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>
#include <unistd.h>

#define N 10

int counter = 0;

sem_t sem_inc;   // 控制 dec 线程
sem_t sem_dec;   // 控制 inc 线程

// 加 1 线程
void* inc(void *arg) {
    int i;
    for (i = 0; i < N; i++) {
        sem_wait(&sem_dec);    // 等待 dec 释放许可

        counter++;             // 临界区：修改共享变量
        printf(" i counter = %d\n", counter);

        sem_post(&sem_inc);    // 唤醒 dec 线程
    }
    pthread_exit(NULL);
}

// 减 1 线程
void* dec(void *arg) {
    int i;
    for (i = 0; i < N; i++) {
        sem_wait(&sem_inc);    // 等待 inc 释放许可

        counter--;             // 临界区：修改共享变量
        printf("d counter = %d\n", counter);

        sem_post(&sem_dec);    // 唤醒 inc 线程
    }
    pthread_exit(NULL);
}

int main() {
    pthread_t tid1, tid2;

    // 初始化信号量
    // sem_inc 初始 0 → dec 必须等待
    // sem_dec 初始 1 → inc 可以先运行
    sem_init(&sem_inc, 0, 0);
    sem_init(&sem_dec, 0, 1);

    // 创建线程
    pthread_create(&tid1, NULL, inc, NULL);
    pthread_create(&tid2, NULL, dec, NULL);

    // 等待线程结束
    pthread_join(tid1, NULL);
    pthread_join(tid2, NULL);

    // 销毁信号量
    sem_destroy(&sem_inc);
    sem_destroy(&sem_dec);

    return 0;
}
```

# 九、macOS 可用版本：GCD dispatch_semaphore
macOS 不推荐 sem_init，用系统原生 **dispatch_semaphore**

```c
#include <stdio.h>
#include <pthread.h>
#include <dispatch/dispatch.h>

#define N 10

int counter = 0;

dispatch_semaphore_t sem_inc;
dispatch_semaphore_t sem_dec;

// 加 1 线程
void* inc(void *arg) {
    for (int i = 0; i < N; i++) {
        // 等待信号量（永久等待）
        dispatch_semaphore_wait(sem_dec, DISPATCH_TIME_FOREVER);

        counter++;
        printf(" i counter = %d\n", counter);

        // 发送信号量
        dispatch_semaphore_signal(sem_inc);
    }
    return NULL;
}

// 减 1 线程
void* dec(void *arg) {
    for (int i = 0; i < N; i++) {
        dispatch_semaphore_wait(sem_inc, DISPATCH_TIME_FOREVER);

        counter--;
        printf("d counter = %d\n", counter);

        dispatch_semaphore_signal(sem_dec);
    }
    return NULL;
}

int main() {
    pthread_t t1, t2;

    // 创建信号量
    sem_inc = dispatch_semaphore_create(0);
    sem_dec = dispatch_semaphore_create(1);

    pthread_create(&t1, NULL, inc, NULL);
    pthread_create(&t2, NULL, dec, NULL);

    pthread_join(t1, NULL);
    pthread_join(t2, NULL);

    // 释放信号量
    dispatch_release(sem_inc);
    dispatch_release(sem_dec);

    return 0;
}
```

# 十、Mutex 与 Semaphore 区别
| 工具 | 作用 |
|------|------|
| **Mutex** | 保护共享变量，互斥访问 |
| **Semaphore** | 控制线程执行顺序、同步 |

---

# 十一、线程核心规则
1. 线程创建用 `pthread_create`
2. 必须用 `pthread_join` 等待线程
3. 共享变量必须用 **Mutex** 保护
4. `sem_wait` 和 `sem_post` 必须成对
5. 信号量初始值决定**谁先运行**
6. macOS 用 `dispatch_semaphore` 代替 `sem_init`
7. 临界区越小，程序效率越高
